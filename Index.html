  // Mobile nav toggle

  const hambtn = document.getElementById('hambtn');

    } else {

      mainNav.style.display = 'flex';

      mainNav.style.flexDirection = 'column';

      mainNav.style.top = '72px';

      mainNav.style.background = 'white';

      mainNav.style.borderRadius = '12px';

    }

  });

  // FAQ toggles

  document.querySelectorAll('.faq-item').forEach(item => {

    item.addEventListener('click', () => {

      const p = item.querySelector('p');

      const open = item.getAttribute('data-open') === 'true';

      if(open){

        p.style.display = 'none';

        item.setAttribute('data-open','false');

      } else {

        p.style.display = 'block';

        item.setAttribute('data-open','true');

      }

    });

  });

  // Simple counters animation (demo numbers)

  function animateCounter(id, target, duration=1200){

    const el = document.getElementById(id);

    let start = 0;

    const stepTime = Math.max(20, duration / target);

    const timer = setInterval(() => {

      start += Math.ceil(target / (duration / stepTime));

      if(start >= target){

        el.textContent = target.toLocaleString();

        clearInterval(timer);

      } else {

        el.textContent = start.toLocaleString();

      }

    }, stepTime);

  }

  // demo values

  animateCounter('c1', 10000, 1200); // 10,000 kg plastics

  animateCounter('c2', 2500, 1200);  // CO2 saved

  animateCounter('c3', 4200, 1200);  // active users

</script>
